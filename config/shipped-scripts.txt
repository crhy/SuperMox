# Scripts shipped by SuperMox packages. Single source of truth: the repo copy
# under scripts/. vm-build-pkg.sh stages each entry into the packaging copy at
# its install path (relative to the package source root), mode 0755.
# Format: <package name> <repo source path> <install path, no leading slash>

# Cluster join CLI and daemon (src/supermox/*.py must be installed where these
# look for them; see bin/supermox-join's sys.path fallback).

supermox-ve bin/supermox-joind usr/sbin/supermox-joind
# Forwards a sysvinit daemon's stderr to syslog (no journald). Used by the
# init.d start lines of daemons whose stderr would otherwise be lost to /dev/null.
supermox-ve scripts/supermox-logpipe usr/lib/supermox/supermox-logpipe
# Home folder sync CLI (init.d/supermox-sync calls it at boot to apply the
