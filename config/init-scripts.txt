#   <component-or-packaging-name> <binary-package> <svc>
# apply-patches.sh copies init.d/<svc> to <out>/<component>/<component>/debian/<binary-package>.<svc>.init

# rrdcached: Debian ships only a systemd unit for it, and upstream pve-cluster
# starts it through Wants=rrdcached.service; without a sysvinit script the RRD
# archives pvestatd files stay empty, so the graphs page has no history. The
# rrdtool Debian source package is forked (patches/rrdtool/), so this script
# ships from the rrdcached package itself and not from supermox-ve: two packages
# owning /etc/init.d/rrdcached would conflict.
